116static void update_volume()
117{
118 int vol;
119
120 /* external input */
121 sceSdSetParam(SD_CORE_1 | SD_PARAM_AVOLL, 0x7fff);
122 sceSdSetParam(SD_CORE_1 | SD_PARAM_AVOLR, 0x7fff);
123
124 /* core0 input */
125 sceSdSetParam(SD_CORE_0 | SD_PARAM_BVOLL, 0);
126 sceSdSetParam(SD_CORE_0 | SD_PARAM_BVOLR, 0);
127
128 /* core1 input */
129 vol = playing ? core1_volume : 0;
130 sceSdSetParam(SD_CORE_1 | SD_PARAM_BVOLL, vol);
131 sceSdSetParam(SD_CORE_1 | SD_PARAM_BVOLR, vol);

133 /* set master volume for core 0 */
134 sceSdSetParam(SD_CORE_0 | SD_PARAM_MVOLL, 0);
135 sceSdSetParam(SD_CORE_0 | SD_PARAM_MVOLR, 0);
136
137 /* set master volume for core 1 */
138 sceSdSetParam(SD_CORE_1 | SD_PARAM_MVOLL, MAX_VOLUME);
139 sceSdSetParam(SD_CORE_1 | SD_PARAM_MVOLR, MAX_VOLUME);
140}

186int audsrv_set_format(int freq, int bits, int channels)
187{
188 int feed_size;
189
190 if (audsrv_format_ok(freq, bits, channels) == 0)
191 {
192 return -AUDSRV_ERR_FORMAT_NOT_SUPPORTED;
193 }
194
195 /* update shift-right count */
196 core1_sample_shift = 0;
197 if (bits == 16)
198 {
199 core1_sample_shift++;
200 }
201
202 if (channels == 2)
203 {
204 core1_sample_shift++;
205 }
206
207 core1_freq = freq;
208 core1_bits = bits;
209 core1_channels = channels;
210
211 /* set ring buffer size to 10 iterations worth of data (512个采样点/48000Hz≈10.7ms，共约107ms) */
212 feed_size = ((512 * core1_freq) / 48000) << core1_sample_shift;
213 ringbuf_size = feed_size * 10;
214
215 /* 缓冲过小的格式(如8000Hz单声道仅1700字节)：放大到20倍feed_size，且不超过静态缓冲上限 */
216 if (ringbuf_size < RINGBUF_ENLARGE_THRESHOLD)
217 {
218 ringbuf_size = feed_size * RINGBUF_LOW_RATE_FEED_COUNT;
219 if (ringbuf_size > (int)sizeof(ringbuf))
220 {
221 ringbuf_size = sizeof(ringbuf);
222 }
223 }
224
225 /* 清空环形缓冲：读位置初始在缓冲中间，播放线程会先读完这半个环，不清空就会先播出上一次残留的旧音频，接缝处产生爆音 */
226 memset(ringbuf, 0, sizeof(ringbuf));
227
228 writepos = 0;
229 /* 初始读位置取缓冲的一半(未放大时恰好是5倍读取量，与原来一致)，保证初始可写空间约半个缓冲 */
230 readpos = (ringbuf_size / 2) & ~3;
231
232 DPRINTF("freq %d bits %d channels %d ringbuf_sz %d feed_size %d shift %d\n", freq, bits, channels, ringbuf_size, feed_size, core1_sample_shift);
233
234 format_changed = 1;
235 return AUDSRV_ERR_NOERROR;
236}

241int audsrv_init()
242{
243 if (initialized)
244 {
245 return 0;
246 }
247
248 /* initialize libsd */
249 if (sceSdInit(SD_INIT_COLD) < 0)
250 {
251 DPRINTF("failed to initialize libsd\n");
252 return -1;
253 }
254
255 readpos = 0;
256 writepos = 0;
257
258 /* initialize transfer_complete's semaphore */
259 transfer_sema = CreateMutex(0);
260 if (transfer_sema < 0)
261 {
262 return AUDSRV_ERR_OUT_OF_MEMORY;
263 }
264
265 queue_sema = CreateMutex(0);
266 if (queue_sema < 0)
267 {
268 DeleteSema(transfer_sema);
269 return AUDSRV_ERR_OUT_OF_MEMORY;
270 }
271
272 /* audio is always playing in the background. trick is to
273 * set the data input volume to zero
274 */
275 audsrv_stop_audio();
276
277 /* initialize transfer-complete callback */
278 sceSdSetTransCallback(AUDSRV_BLOCK_DMA_CH, (void *)transfer_complete);
279 sceSdBlockTrans(AUDSRV_BLOCK_DMA_CH, SD_TRANS_LOOP, core1_buf, sizeof(core1_buf), 0);
280
281 /* default to SPU's native */
282 audsrv_set_format(48000, 16, 2);
283
284 play_tid = create_thread(play_thread, 39, 0);
285 DPRINTF("playing thread 0x%x started\n", play_tid);
286
287 DPRINTF("kickstarted\n");
288
289 initialized = 1;
290 return AUDSRV_ERR_NOERROR;
291}

366int audsrv_play_audio(const char *buf, int buflen)
367{
368 int sent = 0;
369
370 if (initialized == 0)
371 {
372 return -AUDSRV_ERR_NOT_INITIALIZED;
373 }
374
375 if (playing == 0)
376 {
377 /* audio is always playing, just change the volume */
378 playing = 1;
379 update_volume();
380 }
381
382 //DPRINTF("play audio %d bytes, readpos %d, writepos %d avail %d\n", buflen, readpos, writepos, audsrv_available());
383
384 /* limit to what's available, no crossing possible */
385 buflen = MIN(buflen, audsrv_available());
386
387 while (buflen > 0)
388 {
389 int copy = buflen;
390 if (writepos >= readpos)
391 {
392 copy = MIN(ringbuf_size - writepos, buflen);
393 }
394
395 memcpy(ringbuf + writepos, buf, copy);
396 buf = buf + copy;
397 buflen = buflen - copy;
398 sent = sent + copy;
399
400 writepos = writepos + copy;
401 if (writepos >= ringbuf_size)
402 {
403 /* rewind */
404 writepos = 0;
405 }
406 }
407
408 return sent;
409}

451static void play_thread(void *arg)
452{
453 int intr_state;
454 int step;
455 struct upsample_t up;
456 upsampler_t upsampler = NULL;
457
458 (void)arg;
459
460 DPRINTF("starting play thread\n");
461 while (1)
462 {
463 int block;
464 u8 *bufptr;
465 int available;
466
467 if (format_changed)
468 {
469 upsampler = find_upsampler(core1_freq, core1_bits, core1_channels);
470 /* 新格式从头开始，清掉上一首留下的历史采样 */
471 upsampler_reset(&up);
472 format_changed = 0;
473 }
474
475 if (playing && upsampler != NULL)
476 {
477 up.src = (const unsigned char *)ringbuf + readpos;
478 up.left = rendered_left;
479 up.right = rendered_right;
480 step = upsampler(&up);
481
482 readpos = readpos + step;
483 if (readpos >= ringbuf_size)
484 {
485 /* wrap around */
486 readpos = 0;
487 }
488 }
489 else
490 {
491 /* not playing */
492 memset(rendered_left, '\0', sizeof(rendered_left));
493 memset(rendered_right, '\0', sizeof(rendered_right));
494 }
495
496 /* wait until it's safe to transmit another block */
497 WaitSema(transfer_sema);
498
499 /* suspend all interrupts */
500 CpuSuspendIntr(&intr_state);
501
502 /* one block is playing currently, other is idle */
503 block = 1 - (sceSdBlockTransStatus(AUDSRV_BLOCK_DMA_CH, 0) >> 24);
504
505 /* copy 1024 bytes from left and right buffers, into core1_buf */
506 bufptr = core1_buf + (block << 11);
507 wmemcpy(bufptr + 0, rendered_left + 0, 512);
508 wmemcpy(bufptr + 512, rendered_right + 0, 512);
509 wmemcpy(bufptr + 1024, rendered_left + 256, 512);
510 wmemcpy(bufptr + 1536, rendered_right + 256, 512);
511
512 CpuResumeIntr(intr_state);
513
514 available = audsrv_available();
515 if (available >= (ringbuf_size / 10))
516 {
517 /* arbitrarily selected ringbuf_size / 10, to reduce
518 * number of semaphores signalled.
519 */
520 SignalSema(queue_sema);
521 }
522
523 if (fillbuf_threshold > 0 && available >= fillbuf_threshold)
524 {
525 /* EE client requested a callback */
526 call_client_callback(AUDSRV_FILLBUF_CALLBACK);
527 }
528
529 //DPRINTF("avaiable: %d, queued: %d\n", available, ringbuf_size - available);
530 }
531}

536int audsrv_quit()
537{
538 /* silence! */
539 audsrv_stop_audio();
540 audsrv_stop_cd();
541 audsrv_adpcm_init();
542
543 /* stop transmission */
544 sceSdSetTransCallback(AUDSRV_BLOCK_DMA_CH, NULL);
545 sceSdBlockTrans(AUDSRV_BLOCK_DMA_CH, SD_TRANS_STOP, 0, 0, 0);
546
547 /* stop playing thread */
548 if (play_tid > 0)
549 {
550 TerminateThread(play_tid);
551 DeleteThread(play_tid);
552 play_tid = 0;
553 }
554
555#ifndef NO_RPC_THREAD
556 /* Deinitialize RPC client, only once the playback thread is stopped. */
557 deinitialize_rpc_client();
558#endif
559
560 if (transfer_sema > 0)
561 {
562 DeleteSema(transfer_sema);
563 transfer_sema = 0;
564 }
565
566 if (queue_sema > 0)
567 {
568 DeleteSema(queue_sema);
569 queue_sema = 0;
570 }
571
572 return 0;
573}
