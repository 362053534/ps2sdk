35static int mx_sio2_wait_transfer_complete(void)
36{
37 uint32_t timeout = SIO2_STAT_TIMEOUT;
38
39 while ((inl_sio2_stat6c_get() & (1 << 12)) == 0) {
40 if (timeout-- == 0)
41 return 0;
42 }
43
44 return 1;
45}

47/* ISR triggered by the completion of DMA transfer: SIO2 RX FIFO -> MEM */
48int mx_sio2_dma_isr_rx(void *arg)
49{
50 int ef = *(int *)arg;
51 int eflags = EF_SIO2_INTR_REVERSE;
52
53 /* clear SIO2 stat reg */
54 inl_sio2_stat_set(inl_sio2_stat_get());
55
56 /* wait for PIO portion of transfer to complete */
57 if (!mx_sio2_wait_transfer_complete()) {
58 cmd.response = SPISD_RESULT_TIMEOUT;
59 cmd.abort = CMD_ABORT_TRANSFER_TIMEOUT;
60 iSetEventFlag(ef, EF_SIO2_INTR_COMPLETE);
61 return 1;
62 }
63
64#ifdef CONFIG_USE_CRC16
65 /* get CRC16 bytes from RX FIFO */
66 cmd.crc[cmd.sectors_transferred] = reverse_byte_LUT8[inl_sio2_data_in()] << 8;
67 cmd.crc[cmd.sectors_transferred] |= reverse_byte_LUT8[inl_sio2_data_in()];
68#else
69 inl_sio2_data_in();
70 inl_sio2_data_in();
71#endif
72
73 /* NOTE: Some cards respond with 0xFE immediately after the CRC16, others do not.
74 * Try to get 0xFE regardless of whether the transfers complete as it's
75 * needed for correct alignment of CMD12 (STOP_TRANSMISSION) later */
76 if (inl_sio2_data_in() != 0xFE) {
77 cmd.response = mx_sio2_wait_equal(0xFE, READ_TOKEN_TIMEOUT);
78 } else {
79 cmd.response = SPISD_RESULT_OK;
80 }
81
82 cmd.sectors_transferred++;
83
84 if (cmd.sectors_transferred < cmd.sector_count && cmd.abort == 0) {
85
86 if (cmd.response == SPISD_RESULT_OK) {
87 /* start next DMA transfer */
88 mx_sio2_start_rx_dma(&cmd.buffer[cmd.sectors_transferred * SECTOR_SIZE]);
89 /* notify thread data is ready to be reversed */
90 iSetEventFlag(ef, eflags);
91 /* return from ISR */
92 return 1;
93 }
94
95 cmd.abort = CMD_ABORT_NO_READ_TOKEN;
96 }
97
98 /* done or error, notify thread */
99 eflags |= EF_SIO2_INTR_COMPLETE;
100 iSetEventFlag(ef, eflags);
101
102 return 1;
103}

105/* ISR triggered by the completetion of DMA transfer: MEM -> SIO2 TX FIFO */
106int mx_sio2_dma_isr_tx(void *arg)
107{
108 int ef = *(int *)arg;
109 int eflags = 0;
110
111 /* clear SIO2 stat reg */
112 inl_sio2_stat_set(inl_sio2_stat_get());
113
114 /* SIO2 requires all data to be sent to be present in the TX FIFO
115 * prior to starting the transfer. As such, SIO2 will not
116 * have completed the transfer prior to reaching this ISR */
117
118 /* wait for transfer to complete */
119 if (!mx_sio2_wait_transfer_complete()) {
120 cmd.response = SPISD_RESULT_TIMEOUT;
121 cmd.abort = CMD_ABORT_TRANSFER_TIMEOUT;
122 iSetEventFlag(ef, EF_SIO2_INTR_COMPLETE);
123 return 1;
124 }
125
126 /* CRC16 not yet implemented for writes */
127 mx_sio2_write_dummy();
128 mx_sio2_write_dummy();
129
130 /* get data response token */
131 /* 0101 = data accepted */
132 /* 1011 = data rejected, CRC error */
133 /* 1101 = data rejected, write error */
134 cmd.response = mx_sio2_wait_equal_masked(0x5, 0x1F, 10);
135 if (cmd.response != 0x5) {
136 M_DEBUG("ERROR: write data rejected, token 0x%x\n", cmd.response);
137 cmd.abort = CMD_ABORT_WRITE_ERROR;
138 }
139
140 /* wait for card to finish programming */
141 cmd.response = mx_sio2_wait_equal(0xFF, 0x80000);
142 if (cmd.response != SPISD_RESULT_OK) {
143 M_DEBUG("ERROR: failed to finish programming\n");
144 cmd.abort = CMD_ABORT_WRITE_ERROR;
145 }
146
147 if (cmd.abort == 0)
148 cmd.sectors_transferred++;
149
150 if (cmd.sectors_transferred < cmd.sector_count && cmd.abort == 0) {
151 /* send write token */
152 mx_sio2_write_byte(0xFC);
153 /* start next DMA transfer */
154 mx_sio2_start_tx_dma(&cmd.buffer[cmd.sectors_transferred * SECTOR_SIZE]);
155 /* return from ISR */
156 return 1;
157 }
158
159 /* done or error, notify thread */
160 eflags = EF_SIO2_INTR_COMPLETE;
161 iSetEventFlag(ef, eflags);
162
163 return 1;
164}

194void mx_sio2_lock(uint8_t intr_type)
195{
196 int state;
197
198 /* lock sio2man driver so we can use it exclusively */
199 sio2man_hook_sio2_lock();
200
201 /* save ctrl state */
202 sio2_save_crtl = inl_sio2_ctrl_get();
203
204 /* we're in control, setup the ports for our use */
205 mx_sio2_init_ports(&global_td);
206
207 /* enable DMA interrupts */
208 CpuSuspendIntr(&state);
209
210 if (intr_type == INTR_RX) {
211 RegisterIntrHandler(IOP_IRQ_DMA_SIO2_OUT, 1, mx_sio2_dma_isr_rx, &sio2_event_flag);
212 EnableIntr(IOP_IRQ_DMA_SIO2_OUT);
213 }
214
215 if (intr_type == INTR_TX) {
216 RegisterIntrHandler(IOP_IRQ_DMA_SIO2_IN, 1, mx_sio2_dma_isr_tx, &sio2_event_flag);
217 EnableIntr(IOP_IRQ_DMA_SIO2_IN);
218 }
219
220 CpuResumeIntr(state);
221}

223void mx_sio2_unlock(uint8_t intr_type)
224{
225 int state;
226 int res;
227
228 /* disable DMA interrupts */
229 CpuSuspendIntr(&state);
230
231 if (intr_type == INTR_RX) {
232 DisableIntr(IOP_IRQ_DMA_SIO2_OUT, &res);
233 dmac_ch_set_chcr(IOP_DMAC_SIO2out, 0);
234 }
235
236 if (intr_type == INTR_TX) {
237 DisableIntr(IOP_IRQ_DMA_SIO2_IN, &res);
238 dmac_ch_set_chcr(IOP_DMAC_SIO2in, 0);
239 }
240
241 CpuResumeIntr(state);
242
243 if (intr_type == INTR_RX)
244 ReleaseIntrHandler(IOP_IRQ_DMA_SIO2_OUT);
245
246 if (intr_type == INTR_TX)
247 ReleaseIntrHandler(IOP_IRQ_DMA_SIO2_IN);
248
249 /* restore ctrl state, and reset STATE + FIFOS */
250 inl_sio2_ctrl_set(sio2_save_crtl | 0xc);
251
252 /* unlock sio2man driver */
253 sio2man_hook_sio2_unlock();
254}

264uint8_t mx_sio2_write_byte(uint8_t byte)
265{
266 /* reset SIO2 + FIFO pointers, disable interrupts */
267 inl_sio2_ctrl_set(0x0bc);
268
269 inl_sio2_regN_set(0,
270 TR_CTRL_PORT_NR(PORT_NR) |
271 TR_CTRL_PAUSE(0) |
272 TR_CTRL_TX_MODE_PIO_DMA(0) |
273 TR_CTRL_RX_MODE_PIO_DMA(0) |
274 TR_CTRL_NORMAL_TR(1) |
275 TR_CTRL_SPECIAL_TR(0) |
276 TR_CTRL_BAUD_DIV(1) |
277 TR_CTRL_WAIT_ACK_FOREVER(0) |
278 TR_CTRL_TX_DATA_SZ(1) |
279 TR_CTRL_RX_DATA_SZ(1));
280 inl_sio2_regN_set(1, 0);
281
282 /* put byte in TX FIFO */
283 inl_sio2_data_out(reverse_byte_LUT8[byte]);
284
285 /* start queue exec */
286 inl_sio2_ctrl_set(inl_sio2_ctrl_get() | 1);
287
288 /* wait for completion */
289 if (!mx_sio2_wait_transfer_complete())
290 return 0xFF;
291
292#ifdef DEBUG_VERBOSE
293 uint8_t rx = reverse_byte_LUT8[inl_sio2_data_in()];
294 M_DEBUG("W:0x%x, R:0x%x\n", byte, rx);
295 return rx;
296#endif
297
298 /* get byte from RX FIFO */
299 return reverse_byte_LUT8[inl_sio2_data_in()];
300}

341/* 1.440uS delay between bytes with clk div 0x2 */
342uint8_t mx_sio2_wait_equal(uint8_t value, uint32_t count)
343{
344 uint8_t exp_byte = reverse_byte_LUT8[value];
345 uint8_t in_byte = 0;
346
347 while (count > 0 && in_byte != exp_byte) {
348 /* reset SIO2 + FIFO pointers, disable interrupts */
349 inl_sio2_ctrl_set(0x0bc);
350
351 inl_sio2_regN_set(0,
352 TR_CTRL_PORT_NR(PORT_NR) |
353 TR_CTRL_PAUSE(0) |
354 TR_CTRL_TX_MODE_PIO_DMA(0) |
355 TR_CTRL_RX_MODE_PIO_DMA(0) |
356 TR_CTRL_NORMAL_TR(1) |
357 TR_CTRL_SPECIAL_TR(0) |
358 TR_CTRL_BAUD_DIV(1) |
359 TR_CTRL_WAIT_ACK_FOREVER(0) |
360 TR_CTRL_TX_DATA_SZ(0) |
361 TR_CTRL_RX_DATA_SZ(1));
362 inl_sio2_regN_set(1, 0);
363
364 /* start queue exec */
365 inl_sio2_ctrl_set(inl_sio2_ctrl_get() | 1);
366
367 /* wait for completion */
368 if (!mx_sio2_wait_transfer_complete())
369 return SPISD_RESULT_TIMEOUT;
370
371 in_byte = inl_sio2_data_in();
372
373#ifdef DEBUG_VERBOSE
374 M_DEBUG("WE: 0x%x, EX: 0x%x\n", reverse_byte_LUT8[in_byte], value);
375#endif
376
377 count--;
378 }
379
380 return (exp_byte != in_byte) ? SPISD_RESULT_ERROR : SPISD_RESULT_OK;
381}

469{
470 uint32_t transfer_size;
471#ifdef DEBUG_VERBOSE
472 uint8_t *buffer_start = buffer;
473#endif
474 while (size > 0) {
475 /* SIO2 can only transfer 256 bytes at a time */
476 transfer_size = size > SIO2_MAX_TRANSFER_SIZE ? SIO2_MAX_TRANSFER_SIZE : size;
477
478 /* reset SIO2 + FIFO pointers, disable interrupts */
479 inl_sio2_ctrl_set(0x0bc);
480
481 /* add transfer to queue */
482 inl_sio2_regN_set(0,
483 TR_CTRL_PORT_NR(PORT_NR) |
484 TR_CTRL_PAUSE(0) |
485 TR_CTRL_TX_MODE_PIO_DMA(0) |
486 TR_CTRL_RX_MODE_PIO_DMA(0) |
487 TR_CTRL_NORMAL_TR(1) |
488 TR_CTRL_SPECIAL_TR(0) |
489 TR_CTRL_BAUD_DIV(1) |
490 TR_CTRL_WAIT_ACK_FOREVER(0) |
491 TR_CTRL_TX_DATA_SZ(0) |
492 TR_CTRL_RX_DATA_SZ(transfer_size));
493 inl_sio2_regN_set(1, 0);
494
495 /* start queue exec */
496 inl_sio2_ctrl_set(inl_sio2_ctrl_get() | 1);
497
498 /* wait for completion */
499 if (!mx_sio2_wait_transfer_complete())
500 return SPISD_RESULT_TIMEOUT;
501
502 /* PIO: IOP <- SIO2 */
503 for (int i = 0; i < transfer_size; i++) {
504 *buffer++ = reverse_byte_LUT8[inl_sio2_data_in()];
505 }
506
507#ifdef DEBUG_VERBOSE
508 for (int i = 0; i < transfer_size; i++) {
509 M_DEBUG("W:0xFF, R:0x%x\n", buffer_start[i]);
510 }
511#endif
512
513 size -= transfer_size;
514 }
515
516 return SPISD_RESULT_OK;
517}

576void mx_sio2_start_rx_dma(uint8_t *buffer)
577{
578 /* DMA: 256 bytes */
579 const uint32_t tr_ctrl_dma =
580 TR_CTRL_PAUSE(0) |
581 TR_CTRL_TX_MODE_PIO_DMA(0) |
582 TR_CTRL_RX_MODE_PIO_DMA(1) |
583 TR_CTRL_NORMAL_TR(1) |
584 TR_CTRL_SPECIAL_TR(0) |
585 TR_CTRL_TX_DATA_SZ(0) |
586 TR_CTRL_RX_DATA_SZ(0x100) |
587 TR_CTRL_BAUD_DIV(1) |
588 TR_CTRL_WAIT_ACK_FOREVER(0);
589
590 /* PIO: CRC16 (2 bytes) + 0xFE token (1 byte) */
591 const uint32_t tr_ctrl_pio =
592 TR_CTRL_PAUSE(0) |
593 TR_CTRL_TX_MODE_PIO_DMA(0) |
594 TR_CTRL_RX_MODE_PIO_DMA(0) |
595 TR_CTRL_NORMAL_TR(1) |
596 TR_CTRL_SPECIAL_TR(0) |
597 TR_CTRL_TX_DATA_SZ(0) |
598 TR_CTRL_RX_DATA_SZ(3) |
599 TR_CTRL_BAUD_DIV(1) |
600 TR_CTRL_WAIT_ACK_FOREVER(0);
601
602 /* reset SIO2 + FIFO pointers, disable interrupts */
603 inl_sio2_ctrl_set(0x0bc);
604
605 /* add transfers to queue */
606 inl_sio2_regN_set(0, tr_ctrl_dma | TR_CTRL_PORT_NR(PORT_NR));
607 inl_sio2_regN_set(1, tr_ctrl_dma | TR_CTRL_PORT_NR(PORT_NR));
608 inl_sio2_regN_set(2, tr_ctrl_pio | TR_CTRL_PORT_NR(PORT_NR));
609 inl_sio2_regN_set(3, 0);
610
611 /* enable dmac transfer */
612 sceSetSliceDMA(IOP_DMAC_SIO2out, buffer, 0x100 >> 2, 2, DMAC_TO_MEM);
613 sceStartDMA(IOP_DMAC_SIO2out);
614
615 /* start queue exec */
616 inl_sio2_ctrl_set(inl_sio2_ctrl_get() | 1);
617}

637 inl_sio2_regN_set(0, tr_ctrl_dma | TR_CTRL_PORT_NR(PORT_NR));
638 inl_sio2_regN_set(1, tr_ctrl_dma | TR_CTRL_PORT_NR(PORT_NR));
639 inl_sio2_regN_set(2, 0);
640 inl_sio2_regN_set(3, 0);

643 sceSetSliceDMA(IOP_DMAC_SIO2in, buffer, 0x100 >> 2, 2, DMAC_FROM_MEM);

650static void sd_detect()
651{
652 uint16_t results;
653
654 if (!bdm_get_probe_enabled(BDM_PROBE_TYPE_SDC))
655 return;
656
657 mx_sio2_lock(INTR_NONE);
658
659 if (!bdm_get_probe_enabled(BDM_PROBE_TYPE_SDC)) {
660 mx_sio2_unlock(INTR_NONE);
661 return;
662 }
663
664 if (sdcard.initialized == 0) {
665 M_PRINTF("Trying to init card\n");
666 /* bring card up from identification mode to data-transfer mode */
667 results = spisd_init_card();
668 if (results == SPISD_RESULT_OK) {
669 /* get card capacity and attach to BDM */
670 results = spisd_get_card_info();
671 if (results != SPISD_RESULT_OK)
672 results = spisd_recover();
673
674 if (results == SPISD_RESULT_OK) {
675 bdm_connect_bd(&bd);
676 sdcard.initialized = 1;
677 sd_init_retries = 0;
678 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_PRESENT);
679 }
680 }
681
682 if (sdcard.initialized == 0 && sd_init_retries < SD_INIT_MAX_RETRIES) {
683 if (++sd_init_retries == SD_INIT_MAX_RETRIES)
684 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_ERROR);
685 else
686 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_PENDING);
687 }
688 } else {
689 /* try to detect card removal by requesting card status (CMD13) */
690 results = spisd_read_status_register();
691 /* comparing without a mask might be a bit overkill */
692 if (results != 0x0) {
693 /* try to recover */
694 results = spisd_recover();
695 /* maybe add a var to keep track of capacity */
696 if ((sdcard.initialized == 1) && (results != SPISD_RESULT_OK)) {
697 /* recovery failed, disconnect from BDM */
698 M_DEBUG("Recovery failed, disconnecting from bdm.\n");
699 bdm_disconnect_bd(&bd);
700 sdcard.initialized = 0;
701 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_ABSENT);
702 }
703 }
704 }
705
706 mx_sio2_unlock(INTR_NONE);
707}

709static void sd_detect_thread(void *arg)
710{
711 int quickProbeDone = 0;
712 int quickProbePassed = 0;
713
714 (void)arg;
715
716 M_PRINTF("card detection thread running\n");
717
718 while (!sd_detect_thread_stop) {
719 /* try to detect card removal if it hasn't been used recently */
720 if (sdcard.used == 0 && bdm_get_probe_enabled(BDM_PROBE_TYPE_SDC)) {
721 if (!sdcard.initialized && !quickProbeDone) {
722 // 前1秒每帧进行一次极简探测，成功后立即进入原初始化流程。
723 for (int i = 0; i < 60 && !sd_detect_thread_stop && bdm_get_probe_enabled(BDM_PROBE_TYPE_SDC); i++) {
724 int result;
725
726 mx_sio2_lock(INTR_NONE);
727 result = spisd_probe_card(i == 0);
728 mx_sio2_unlock(INTR_NONE);
729
730 if (result == SPISD_RESULT_OK) {
731 quickProbePassed = 1;
732 break;
733 }
734
735 DelayThread(16667);
736 }
737 quickProbeDone = 1;
738 if (!quickProbePassed) {
739 sd_init_retries = 0;
740 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_ABSENT);
741 }
742 } else if (!sdcard.initialized && !quickProbePassed) {
743 mx_sio2_lock(INTR_NONE);
744 quickProbePassed = spisd_probe_card(0) == SPISD_RESULT_OK;
745 mx_sio2_unlock(INTR_NONE);
746
747 if (quickProbePassed) {
748 sd_init_retries = 0;
749 bdm_set_probe_state(BDM_PROBE_TYPE_SDC, BDM_PROBE_STATE_PENDING);
750 }
751 }
752
753 if (sdcard.initialized || quickProbePassed) {
754 int initialized = sdcard.initialized;
755
756 sd_detect();
757 // 游戏运行期间不再需要周期检测，初始化成功后结束检测线程。
758 if (game_mode && sdcard.initialized)
759 break;
760
761 if (initialized && !sdcard.initialized) {
762 quickProbeDone = 0;
763 quickProbePassed = 0;
764 }
765 }
766 } else if (!bdm_get_probe_enabled(BDM_PROBE_TYPE_SDC)) {
767 quickProbeDone = 0;
768 quickProbePassed = 0;
769 }
770 sdcard.used = 0;
771
772 DelayThread(1000 * 1000);
773 }
774
775 ExitDeleteThread();
776}

780/* 58-59uS avg on DECKARD */
781#pragma GCC push_options
782#pragma GCC optimize("-O3")
783inline void reverse_buffer(uint32_t *buffer, uint32_t count)
784{
785 const uint32_t mask0F = 0x0F0F0F0F;
786 const uint32_t maskF0 = 0xF0F0F0F0;
787 const uint32_t mask33 = 0x33333333;
788 const uint32_t maskCC = 0xCCCCCCCC;
789 const uint32_t mask55 = 0x55555555;
790 const uint32_t maskAA = 0xAAAAAAAA;
791 uint32_t n;
792
793#pragma GCC unroll 2
794 for (int i = 0; i < count; i++) {
795 n = buffer[i];
796 n = ((n & maskF0) >> 4) | ((n & mask0F) << 4);
797 n = ((n & maskCC) >> 2) | ((n & mask33) << 2);
798 n = ((n & maskAA) >> 1) | ((n & mask55) << 1);
799
800 buffer[i] = n;
801 }
802}
