29static int bd_defrag_locate(struct block_device *bd, u32 fragcount, struct bd_fragment *fraglist, u64 sector, u32 *fragment, u64 *offset, bd_defrag_cursor_t *cursor)
30{
31 u32 i = 0;
32 u64 current_offset = 0;
33
34 /* 顺序读取时从上次命中的碎片继续查找，随机回跳时退回表头。 */
35 if (cursor != NULL && cursor->bd == bd && cursor->fraglist == fraglist && cursor->fragcount == fragcount && cursor->fragment < fragcount) {
36 struct bd_fragment *cached = &fraglist[cursor->fragment];
37
38 if (cursor->offset <= sector) {
39 i = cursor->fragment;
40 current_offset = cursor->offset;
41 if (sector < current_offset + cached->count) {
42 *fragment = i;
43 *offset = current_offset;
44 return 0;
45 }
46
47 current_offset += cached->count;
48 i++;
49 }
50 }
51
52 for (; i < fragcount; i++) {
53 struct bd_fragment *f = &fraglist[i];
54
55 if (current_offset <= sector && sector < current_offset + f->count) {
56 *fragment = i;
57 *offset = current_offset;
58 return 0;
59 }
60 current_offset += f->count;
61 }
62
63 bd_defrag_cursor_reset(cursor);
64 return -1;
65}

67int bd_defrag_read_cached(struct block_device *bd, u32 fragcount, struct bd_fragment *fraglist, u64 sector, void *buffer, u16 count, bd_defrag_cursor_t *cursor)
68{
69 bd_defrag_cursor_t local_cursor;
70 u64 sector_start = sector;
71 u16 count_left = count;
72
73 if (count == 0)
74 return 0;
75
76 /* 连续文件保持一次定位、一次底层读取，不进入多碎片路径。 */
77 if (fragcount == 1 && sector < fraglist[0].count && count <= fraglist[0].count - sector) {
78 bd_defrag_cursor_set(cursor, bd, fragcount, fraglist, 0, 0);
79 if (bd->read(bd, fraglist[0].sector + sector, buffer, count) != count) {
80 M_PRINTF("%s: ERROR: read failed!\n", __FUNCTION__);
81 return -1;
82 }
83 return count;
84 }
85
86 if (cursor == NULL) {
87 bd_defrag_cursor_reset(&local_cursor);
88 cursor = &local_cursor;
89 }
90
91 while (count_left > 0) {
92 u16 count_read;
93 u64 offset;
94 u64 readable;
95 u64 physical_end;
96 struct bd_fragment *f;
97 u32 fragment;
98 u32 last_fragment;
99 u64 last_offset;
100
101 if (bd_defrag_locate(bd, fragcount, fraglist, sector_start, &fragment, &offset, cursor) != 0) {
102 M_PRINTF("%s: ERROR: fragment not found!\n", __FUNCTION__);
103 return -1;
104 }
105
106 f = &fraglist[fragment];
107 readable = offset + f->count - sector_start;
108 physical_end = f->sector + f->count;
109 last_fragment = fragment;
110 last_offset = offset;
111
112 /* 物理地址连续的相邻项可以合成同一条底层读取命令。 */
113 while (readable < count_left && last_fragment + 1 < fragcount) {
114 struct bd_fragment *next = &fraglist[last_fragment + 1];
115
116 if (next->count == 0 || next->sector != physical_end)
117 break;
118
119 last_offset += fraglist[last_fragment].count;
120 readable += next->count;
121 physical_end += next->count;
122 last_fragment++;
123 }
124
125 count_read = count_left;
126 if (count_read > readable) {
127 count_read = readable;
128 M_DEBUG("%s: clipping sectors %d -> %d\n", __FUNCTION__, count_left, count_read);
129 }
130
131 if (bd->read(bd, f->sector + (sector_start - offset), buffer, count_read) != count_read) {
132 M_PRINTF("%s: ERROR: read failed!\n", __FUNCTION__);
133 return -1;
134 }
135
136 bd_defrag_cursor_set(cursor, bd, fragcount, fraglist, last_fragment, last_offset);
137 sector_start += count_read;
138 count_left -= count_read;
139 buffer = (u8 *)buffer + (count_read * bd->sectorSize);
140 }
141
142 return count;
143}

150int bd_defrag_write(struct block_device* bd, u32 fragcount, struct bd_fragment* fraglist, u64 sector, const void* buffer, u16 count)
151{
152 u64 sector_start = sector;
153 u16 count_left = count;
154
155 while (count_left > 0) {
156 u16 count_write;
157 u64 offset = 0; // offset of fragment in bd/file
158 struct bd_fragment *f = NULL;
159 int i;
160
161 // Locate fragment containing start sector
162 for (i=0; (u32)i<fragcount; i++) {
163 f = &fraglist[i];
164 if (offset <= sector_start && (offset + f->count) > sector_start) {
165 // Fragment found
166 break;
167 }
168 offset += f->count;
169 }
170
171 if ((u32)i == fragcount) {
172 M_PRINTF("%s: ERROR: fragment not found!\n", __FUNCTION__);
173 return -1;
174 }
175
176 // Clip to fragment size
177 count_write = count_left;
178 if ((sector_start + count_write) > (offset + f->count)) {
179 count_write = (offset + f->count) - sector_start;
180 M_DEBUG("%s: clipping sectors %d -> %d\n", __FUNCTION__, count_left, count_write);
181 }
182
183 // Do the write
184 if (bd->write(bd, f->sector + (sector_start - offset), buffer, count_write) != count_write) {
185 M_PRINTF("%s: ERROR: write failed!\n", __FUNCTION__);
186 return -1;
187 }
188
189 // Advance to next fragment
190 sector_start += count_write;
191 count_left -= count_write;
192 buffer = (u8*)buffer + (count_write * bd->sectorSize);
193 }
194
195 return count;
196}
