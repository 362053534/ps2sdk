103static void up_generic_16_stereo(struct upsample_t *up, const short *lut)
104{
105 int p;
106 short *left = up->left;
107 short *right = up->right;
108 short *src = (short *)up->src;
109
110 for (p=0; p<512; p++)
111 {
112 int q;
113
114 q = lut[p] << 1;
115 *left++ = src[q + 0];
116 *right++ = src[q + 1];
117 }
118}

138#define UP_OUTPUT_COUNT 512 /* 每块输出的采样点数 */
139#define UP_CUBIC_HISTORY UP_HISTORY_SAMPLES /* 插值窗口比当前块多需要的历史采样数，定义在头文件 */
140#define UP_CUBIC_DELAY 2 /* 输出相对输入的延后(输入采样数)，保证窗口最远只读到本块最后一个采样 */
141#define UP_POS_SHIFT 7 /* 位置的16.16定点小数每个输出点的步进 = step << 7 (65536 / 512 = 128) */
142#define UP_FRAC_BITS 15 /* 插值小数部分的位数 */
143#define UP_FRAC_ONE (1 << UP_FRAC_BITS)
144#define UP_WEIGHT_BITS 14 /* 权重的定点位数(16位采样 x 权重 的和不会超过32位) */
145#define UP_STEP_MAX 512 /* 每块最多消耗的输入帧数，大于它的采样率不走插值 */

185static void up_cubic_build(up_cubic_table_t *t, int step)
186{
187 int i;
188
189 for (i = 0; i < UP_OUTPUT_COUNT; i++)
190 {
191 int pos, j, f, f2, f3, w0, w1, w2, w3;
192
193 /* 输出点i在输入中的位置(16.16定点)，减去延后量 */
194 pos = ((i * step) << UP_POS_SHIFT) - (UP_CUBIC_DELAY << 16);
195 j = pos >> 16;
196 f = (pos & 0xffff) >> (16 - UP_FRAC_BITS);
197 f2 = (f * f) >> UP_FRAC_BITS;
198 f3 = (f2 * f) >> UP_FRAC_BITS;
199
200 /* Catmull-Rom 权重(15位定点)，4个权重之和为 1.0 */
201 w0 = (-f3 + 2 * f2 - f) >> 1;
202 w1 = (3 * f3 - 5 * f2 + 2 * UP_FRAC_ONE) >> 1;
203 w2 = (-3 * f3 + 4 * f2 + f) >> 1;
204 w3 = (f3 - f2) >> 1;
205
206 /* 窗口起点：j-1，再加上工作缓冲里历史采样占的位置 */
207 t->base[i] = (short)(j - 1 + UP_CUBIC_HISTORY);
208
209 /* 降到14位存储(15位的1.0放不进short) */
210 t->weight[i][0] = (short)((w0 + 1) >> (UP_FRAC_BITS - UP_WEIGHT_BITS));
211 t->weight[i][1] = (short)((w1 + 1) >> (UP_FRAC_BITS - UP_WEIGHT_BITS));
212 t->weight[i][2] = (short)((w2 + 1) >> (UP_FRAC_BITS - UP_WEIGHT_BITS));
213 t->weight[i][3] = (short)((w3 + 1) >> (UP_FRAC_BITS - UP_WEIGHT_BITS));
214 }
215
216 t->step = step;
217}

226static void up_cubic_render(const up_cubic_table_t *t, int step, const short *work, short *history, short *out)
227{
228 int i;
229
230 for (i = 0; i < UP_OUTPUT_COUNT; i++)
231 {
232 const short *p = work + t->base[i];
233 const short *w = t->weight[i];
234 int v;
235
236 v = p[0] * w[0] + p[1] * w[1] + p[2] * w[2] + p[3] * w[3];
237 v = (v + (1 << (UP_WEIGHT_BITS - 1))) >> UP_WEIGHT_BITS;
238
239 /* 三次插值会略微过冲，限幅到16位范围 */
240 if (v > 32767)
241 {
242 v = 32767;
243 }
244 else if (v < -32768)
245 {
246 v = -32768;
247 }
248
249 out[i] = (short)v;
250 }
251
252 for (i = 0; i < UP_CUBIC_HISTORY; i++)
253 {
254 history[i] = work[step + i];
255 }
256}

272static int up_cubic(struct upsample_t *up, up_cubic_table_t *t, int step, int bits, int channels)
273{
274 int i, k;
275
276 if (t->step != step)
277 {
278 up_cubic_build(t, step);
279 }
280
281 for (i = 0; i < UP_CUBIC_HISTORY; i++)
282 {
283 up_work_left[i] = up->history_left[i];
284 up_work_right[i] = up->history_right[i];
285 }
286
287 if (bits == 16)
288 {
289 const short *src = (const short *)up->src;
290
291 for (k = 0; k < step; k++)
292 {
293 if (channels == 2)
294 {
295 up_work_left[UP_CUBIC_HISTORY + k] = src[2 * k + 0];
296 up_work_right[UP_CUBIC_HISTORY + k] = src[2 * k + 1];
297 }
298 else
299 {
300 up_work_left[UP_CUBIC_HISTORY + k] = src[k];
301 up_work_right[UP_CUBIC_HISTORY + k] = src[k];
302 }
303 }
304 }
305 else
306 {
307 const signed char *src = (const signed char *)up->src;
308
309 for (k = 0; k < step; k++)
310 {
311 if (channels == 2)
312 {
313 up_work_left[UP_CUBIC_HISTORY + k] = up_convert_8(src[2 * k + 0]);
314 up_work_right[UP_CUBIC_HISTORY + k] = up_convert_8(src[2 * k + 1]);
315 }
316 else
317 {
318 up_work_left[UP_CUBIC_HISTORY + k] = up_convert_8(src[k]);
319 up_work_right[UP_CUBIC_HISTORY + k] = up_work_left[UP_CUBIC_HISTORY + k];
320 }
321 }
322 }
323
324 up_cubic_render(t, step, up_work_left, up->history_left, up->left);
325 if (channels == 2)
326 {
327 up_cubic_render(t, step, up_work_right, up->history_right, up->right);
328 }
329 else
330 {
331 /* 单声道：右声道与左声道相同 */
332 for (i = 0; i < UP_OUTPUT_COUNT; i++)
333 {
334 up->right[i] = up->left[i];
335 }
336 }
337
338 return step * (bits / 8) * channels;
339}

495upsampler_t find_upsampler(int freq, int bits, int channels)
496{
497 struct entry_t *p = upsamplers;
498 while (p->func != NULL)
499 {
500 if (p->freq == freq && p->bits == bits && p->channels == channels)
501 {
502 /* found us an upsampler */
503 return p->func;
504 }
505
506 p++;
507 }
508
509 /* no more upsamplers */
510 return NULL;
511}
